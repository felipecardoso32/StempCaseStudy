//
//  EventStore.swift
//  StempCaseStudy
//
//  Substitui o Airtable + WebService + EventsViewModel do app original.
//  Fonte única da verdade para as telas "Desafios", "Dashboard" e "Eventos"
//  e, via EventStorage (App Group), para o Widget.
//
import SwiftUI
import WidgetKit

final class EventStore: ObservableObject {
    @Published private(set) var events: [EventDetail] = []

    private let seenKey = "seenIdEvents"

    init() {
        events = EventStorage.loadEvents()
    }

    // MARK: - Leitura

    /// Eventos de hoje em diante, ordenados (mesma regra do Airtable original).
    var upcomingEvents: [EventDetail] {
        EventStorage.upcoming(events)
    }

    var nextEvent: EventDetail? {
        upcomingEvents.first
    }

    var newEventsCount: Int {
        let seen = seenIds
        return upcomingEvents.filter { $0.time > Date() && !seen.contains($0.id) }.count
    }

    // MARK: - Escrita

    func add(_ event: EventDetail) {
        events.append(event)
        persist()
    }

    func remove(id: String) {
        events.removeAll { $0.id == id }
        persist()
    }

    func removeAll() {
        events.removeAll()
        persist()
    }

    func loadSamples() {
        events.append(contentsOf: SampleData.events())
        persist()
    }

    // MARK: - "Novo" (igual ao EventsViewModel original)

    private var seenIds: Set<String> {
        get { Set(UserDefaults.standard.stringArray(forKey: seenKey) ?? []) }
        set { UserDefaults.standard.set(Array(newValue), forKey: seenKey) }
    }

    func isNew(_ event: EventDetail) -> Bool {
        !seenIds.contains(event.id)
    }

    func markAsSeen(_ event: EventDetail) {
        seenIds.insert(event.id)
        objectWillChange.send()
    }

    // MARK: - Persistência

    private func persist() {
        EventStorage.saveEvents(events)
        let currentIds = Set(events.map(\.id))
        seenIds = seenIds.intersection(currentIds)
        WidgetCenter.shared.reloadAllTimelines()
    }
}

//
//  EventStorage.swift
//  StempCaseStudy  ·  Shared (App + Widget)
//
//  Camada de persistência compartilhada via App Group.
//  O app grava aqui; o Widget (WidgetKit) lê daqui.
//
//  ⚠️ Troque `appGroupID` pelo App Group que você criar em
//  Signing & Capabilities (no target do App E no target do Widget).
//  Se o App Group não existir, cai no UserDefaults.standard
//  (o app funciona, mas o Widget não enxerga os dados).
//
import Foundation

enum EventStorage {
    static let appGroupID = "group.com.example.stempcasestudy"
    static let eventsKey = "cachedEvents"

    static var defaults: UserDefaults {
        UserDefaults(suiteName: appGroupID) ?? .standard
    }

    private static var encoder: JSONEncoder {
        let e = JSONEncoder()
        e.dateEncodingStrategy = .iso8601
        return e
    }

    private static var decoder: JSONDecoder {
        let d = JSONDecoder()
        d.dateDecodingStrategy = .iso8601
        return d
    }

    static func loadEvents() -> [EventDetail] {
        guard let data = defaults.data(forKey: eventsKey),
              let events = try? decoder.decode([EventDetail].self, from: data)
        else { return [] }
        return events
    }

    static func saveEvents(_ events: [EventDetail]) {
        guard let data = try? encoder.encode(events) else { return }
        defaults.set(data, forKey: eventsKey)
    }

    /// Mesma regra do Airtable original: `{Datetime} >= TODAY()`, ordenado por data crescente.
    static func upcoming(_ events: [EventDetail], from now: Date = Date()) -> [EventDetail] {
        let startOfToday = Calendar.current.startOfDay(for: now)
        return events
            .filter { $0.time >= startOfToday }
            .sorted { $0.time < $1.time }
    }
}

import SwiftUI

extension Color {
    init(hex: String) {
        let hex = hex.trimmingCharacters(in: CharacterSet.alphanumerics.inverted)
        var int: UInt64 = 0
        Scanner(string: hex).scanHexInt64(&int)
        let r = Double((int >> 16) & 0xFF) / 255
        let g = Double((int >> 8) & 0xFF) / 255
        let b = Double(int & 0xFF) / 255
        self.init(red: r, green: g, blue: b)
    }
}

//
//  EventDetail.swift
//  StempCaseStudy  ·  Shared (App + Widget)
//
//  Mesmo modelo do app original (EventDetail), mas sem Airtable:
//  o que antes era uma string "Título - dd/MM/yyyy - link, ..." agora é
//  uma lista estruturada de entregáveis (mais fácil de usar no Widget).
//
import Foundation

struct SupportMaterial: Identifiable, Codable, Hashable {
    var id: String = UUID().uuidString
    var filename: String = ""
    var url: String = ""
}

struct Deliverable: Identifiable, Codable, Hashable {
    var id: String = UUID().uuidString
    var title: String = ""
    var dueDate: Date = Calendar.current.date(byAdding: .day, value: 7, to: Date()) ?? Date()
    var url: String = ""

    var formattedDueDate: String {
        let fmt = DateFormatter()
        fmt.locale = Locale(identifier: "pt_BR")
        fmt.dateFormat = "dd/MM/yyyy"
        return fmt.string(from: dueDate)
    }
}

struct EventDetail: Identifiable, Codable, Hashable {
    let id: String
    let name: String
    let typeEvent: String          // rawValue de EventKind
    let time: Date
    let icon: String               // SF Symbol
    let colorHex: String
    let description: String?
    let supportMaterial: [SupportMaterial]
    let deliverables: [Deliverable]

    init(
        id: String = UUID().uuidString,
        name: String,
        typeEvent: String,
        time: Date,
        icon: String,
        colorHex: String,
        description: String? = nil,
        supportMaterial: [SupportMaterial] = [],
        deliverables: [Deliverable] = []
    ) {
        self.id = id
        self.name = name
        self.typeEvent = typeEvent
        self.time = time
        self.icon = icon
        self.colorHex = colorHex
        self.description = description
        self.supportMaterial = supportMaterial
        self.deliverables = deliverables
    }

    var formattedDateTime: String {
        let fmt = DateFormatter()
        fmt.locale = Locale(identifier: "pt_BR")
        fmt.dateFormat = "d 'de' MMM 'às' HH:mm"
        return fmt.string(from: time)
    }

    var formattedTime: String {
        let fmt = DateFormatter()
        fmt.locale = Locale(identifier: "pt_BR")
        fmt.dateFormat = "HH:mm"
        return fmt.string(from: time)
    }

    /// Próximo prazo de entrega ainda não vencido (útil para o Widget).
    var nextDeliverable: Deliverable? {
        deliverables
            .filter { $0.dueDate >= Calendar.current.startOfDay(for: Date()) }
            .sorted { $0.dueDate < $1.dueDate }
            .first
    }
}

            .filter { !$0.filename.trimmingCharacters(in: .whitespaces).isEmpty }

    @State private var animatedDelays: Double = 0
    @State private var animatedAbsences: Double = 0

                VStack(alignment: .leading, spacing: 3) {
                    Text("Frequência")
                        .font(.subheadline).fontWeight(.medium)

                        Text("\(start.formatted(.dateTime.day().month(.abbreviated).year())) — Hoje")

            VStack(spacing: 20) {
                ForEach(rows, id: \.0) { label, animated, color in
                    VStack(spacing: 7) {
                        HStack(alignment: .lastTextBaseline) {
                            Text(label)
                                .font(.subheadline)
                                .foregroundColor(.white.opacity(0.55))
                            Spacer()
                            Text(total > 0 ? "\(Int(pct(animated) * 100))%" : "0%")
                                .font(.caption)
                                .foregroundColor(.white.opacity(0.35))
                        }
                        GeometryReader { geo in
                            ZStack(alignment: .leading) {
                                RoundedRectangle(cornerRadius: 999)
                                    .fill(Color.white.opacity(0.07))
                                    .frame(height: 10)
                                RoundedRectangle(cornerRadius: 999)
                                    .fill(color)
                                    .frame(width: geo.size.width * pct(animated), height: 10)
                            }
                        }
                        .frame(height: 10)
                    }
                }
            }

            Divider().overlay(Color.white.opacity(0.07))

            HStack(spacing: 8) {
                ForEach(Array([
                    ("Presenças", presences, Color(hex: "#1D9E75")),
                    ("Atrasos", delays, Color(hex: "#EF9F27")),
                    ("Faltas", absences, Color(hex: "#D85A30")),
                    ("Total", total, Color.white.opacity(0.6))
                ].enumerated()), id: \.offset) { index, item in
                    VStack(spacing: 3) {
                        Text("\(item.1)")
                            .font(.system(size: 26, weight: .semibold))
                            .foregroundColor(item.2)
                        Text(item.0)
                            .font(.system(size: 11))
                            .foregroundColor(.white.opacity(0.35))
                    }
                    .frame(maxWidth: .infinity)
                    .padding(.vertical, 14)
                    .background(Color.white.opacity(0.05))
                    .cornerRadius(10)
                    .opacity(cardsVisible ? 1 : 0)
                    .offset(y: cardsVisible ? 0 : 8)
                    .animation(
                        .easeOut(duration: 0.5).delay(0.7 + Double(index) * 0.12),
                        value: cardsVisible
                    )
                }
            }
        }
        .padding()
        .cornerRadius(16)
        .onAppear {
            withAnimation(.easeInOut(duration: 1.1)) {
                animatedPresences = Double(presences)
                animatedDelays = Double(delays)
                animatedAbsences = Double(absences)
            }
            cardsVisible = true
        }
    }
}
